52/**
53 * EM mapping from physical core ids <-> logical core ids
54 */
55typedef struct {
56 /**
57 * Number of EM-cores currently running (em_init_local/core() called).
58 * Includes EM worker and control cores, but not external threads.
59 */
60 odp_atomic_u32_t current_core_count;
61 /** Number of EM worker cores currently running */
62 odp_atomic_u32_t worker_core_count;
63 /** Number of EM control cores currently running */
64 odp_atomic_u32_t control_core_count;
65 /** Number of EM external threads currently running */
66 odp_atomic_u32_t external_thread_count;
67
68 /** Number of ongoing TERM_THREAD_REQ operations per thread */
69 odp_atomic_u32_t ongoing_term_reqs[ODP_THREAD_COUNT_MAX];
70
71 /**
72 * Read-write lock for the 'rwlocked'-struct below.
73 *
74 * Multiple readers of core-map data are allowed at the same time,
75 * but only one writer. The only two scenarios where the lock is taken
76 * for writes are adding an EM-core via em_init_local/core() and
77 * removing an EM-core via em_term_local/core() - thus taking the lock
78 * for reads in control operations is not time crtitical since add/rem
79 * core operations can afford to wait.
80 *
81 * This lock is also used to block EM-core adds or rems when an internal
82 * control operation is under way that needs info about the cores
83 * available and sends control events to them - cores can not be allowed
84 * to be removed in the middle of this.
85 */
86 odp_rwlock_t rwlock;
87 struct {
88 /*
89 * ODP-thread-id <-> EM-core mapping available for EM-cores
90 * of type EM_CORE_TYPE_WORKER and EM_CORE_TYPE_CONTROL.
91 */
92 struct {
93 /** From ODP thread ids to logical EM core ids */
94 uint8_t logic[ODP_THREAD_COUNT_MAX];
95 /** From logical EM core ids to ODP thread ids */
96 uint16_t odp_thr[EM_MAX_CORES];
97 } thr_vs_logic;
98
99 /** Mask of logic EM-core ids (worker and control EM-cores) */
100 em_core_mask_t logic_mask;
101 /** Mask of logic worker EM-core ids */
102 em_core_mask_t worker_mask;
103 /** Mask of logic control EM-core ids */
104 em_core_mask_t control_mask;
105
106 /** Mask of ODP-thread ids of worker _and_ control EM-cores */
107 odp_thrmask_t thrmask_emcores;
108 /** Mask of ODP-thread ids of EM external threads */
109 odp_thrmask_t thrmask_extthrs;
110 /** Mask of ODP-thread ids of all threads */
111 odp_thrmask_t thrmask_all;
112
113 /** Thread ids of EM-cores (worker and control) */
114 pid_t tids[EM_MAX_CORES];
115 /** Thread ids of external EM threads/processes */
116 pid_t ext_tids[ODP_THREAD_COUNT_MAX];
117
118 /** Type of each EM-core: worker or ctrl (not external) */
119 em_core_type_t core_types[EM_MAX_CORES];
120
121 /**
122 * Store file paths to each EM-core's /proc/[tid]/stat file
123 * so that the paths don't have to be generated each time.
124 */
125 char proc_tid_stat_paths[EM_MAX_CORES][32];
126 } rwlocked;
127
128 /* Pad size to a multiple of cache line size */
129 void *end[0] ENV_CACHE_LINE_ALIGNED;
130} core_map_t;
131
132COMPILE_TIME_ASSERT((sizeof(core_map_t) % ENV_CACHE_LINE_SIZE) == 0,
133 CORE_MAP_T__SIZE_ERROR);
134COMPILE_TIME_ASSERT(EM_MAX_CORES - 1 <= UINT8_MAX,
135 CORE_MAP_T__TYPE_ERROR1);
136COMPILE_TIME_ASSERT(ODP_THREAD_COUNT_MAX - 1 <= UINT16_MAX,
137 CORE_MAP_T__TYPE_ERROR2);
138
139#ifdef __cplusplus
140}
141#endif
