69/**
70 * EM queue group element
71 */
72typedef struct queue_group_elem_t {
73 /** EM queue group handle/id */
74 em_queue_group_t queue_group;
75 /** Associated ODP schedule group */
76 odp_schedule_group_t odp_sched_group;
77 /** Number of queues that belong to this queue group */
78 odp_atomic_u32_t num_queues;
79 /** Queue list, all queues that belong to this queue group */
80 list_node_t queue_list;
81 /** EM core mask */
82 em_core_mask_t core_mask;
83 union {
84 /**
85 * Queue group pool elem for linking free queue groups for
86 * queue_group_alloc().
87 * Used when the queue group is not yet allocated (i.e. is in
88 * the qgrp pool).
89 */
90 objpool_elem_t queue_group_pool_elem;
91 /**
92 * Queue group elements to be flushed in EM-core termination.
93 * Used as a node when creating a list of all queue groups with
94 * only the terminating core set in the mask - the queues
95 * belonging to these qgrps should be flushed before terminating
96 * the core.
97 */
98 list_node_t qgrp_list_node;
99 };
100 /** Is queue group deletion ongoing? true/false */
101 bool ongoing_delete;
102 /** Queue group elem lock */
103 odp_ticketlock_t lock;
104 /** Pad to multiple of cache line size */
105 void *end[0] ENV_CACHE_LINE_ALIGNED;
106} queue_group_elem_t ENV_CACHE_LINE_ALIGNED;
107
108COMPILE_TIME_ASSERT(sizeof(queue_group_elem_t) % ENV_CACHE_LINE_SIZE == 0,
109 QUEUE_GROUP_ELEM_T__SIZE_ERROR);
