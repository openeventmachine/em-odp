52/*
53 * Atomic operations - types & functions
54 */
55struct _env_atomic32 {
56 uint32_t a32; /**< Storage for the atomic variable */
57} ENV_ALIGNED(sizeof(uint32_t)); /* Enforce alignment! */
58
59struct _env_atomic64 {
60 uint64_t a64; /**< Storage for the atomic variable */
61} ENV_ALIGNED(sizeof(uint64_t)); /* Enforce alignment! */
62
63typedef struct _env_atomic32 env_atomic32_t;
64typedef struct _env_atomic64 env_atomic64_t;
65
66/*
67 * 32 bit functions
68 */
69static inline void
70env_atomic32_init(env_atomic32_t *atom)
71{
72 __atomic_store_n(&atom->a32, 0, __ATOMIC_SEQ_CST);
73}
74
75static inline void
76env_atomic32_set(env_atomic32_t *atom, uint32_t new_val)
77{
78 __atomic_store_n(&atom->a32, new_val, __ATOMIC_SEQ_CST);
79}
80
81static inline uint32_t
82env_atomic32_get(const env_atomic32_t *atom)
83{
84 return __atomic_load_n(&atom->a32, __ATOMIC_SEQ_CST);
85}
86
87static inline void
88env_atomic32_dec(env_atomic32_t *atom)
89{
90 (void)__atomic_fetch_sub(&atom->a32, 1, __ATOMIC_SEQ_CST);
91}
92
93static inline void
94env_atomic32_inc(env_atomic32_t *atom)
95{
96 (void)__atomic_fetch_add(&atom->a32, 1, __ATOMIC_SEQ_CST);
97}

129static inline uint32_t
130env_atomic32_return_sub(env_atomic32_t *atom, uint32_t sub_val)

135static inline int
136env_atomic32_cmpset(env_atomic32_t *atom, uint32_t expected, uint32_t desired)
137{
138 return __atomic_compare_exchange_n(&atom->a32, &expected, desired,
139 0 /*strong*/, __ATOMIC_SEQ_CST,
140 __ATOMIC_SEQ_CST);
141}
142
143static inline uint32_t
144env_atomic32_exchange(env_atomic32_t *atom, uint32_t new_val)
145{
146 return __atomic_exchange_n(&atom->a32, new_val, __ATOMIC_SEQ_CST);
147}
148
149static inline void
150env_atomic32_set_bits(env_atomic32_t *atom, uint32_t bit_mask)
151{
152 uint32_t old_val;
153 uint32_t new_val;
154 int ret;
155
156 do {
157 old_val = env_atomic32_get(atom);
158 new_val = old_val | bit_mask;
159 ret = env_atomic32_cmpset(atom, old_val, new_val);
160 } while (ret == 0);
161}
162
163static inline void
164env_atomic32_clr_bits(env_atomic32_t *atom, uint32_t bit_mask)
165{
166 uint32_t old_val;
167 uint32_t new_val;
168 int ret;
169
170 do {
171 old_val = env_atomic32_get(atom);
172 new_val = old_val & (~bit_mask);
173 ret = env_atomic32_cmpset(atom, old_val, new_val);
174 } while (ret == 0);
175}
176
177static inline int
178env_atomic32_cnt_bits(const env_atomic32_t *atom)
179{
180 return __builtin_popcount(env_atomic32_get(atom));
181}
