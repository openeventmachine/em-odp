36/**
37 * @file
38 * @defgroup em_error Error management
39 * Error/Exception management
40 *
41 * EM provides two basic mechanisms to manage operational exceptions
42 *
43 * -# Return value
44 * - Most API calls return a status which can be checked by the application
45 *
46 * -# Error handler
47 * - An optional error handler function can be registered. The given
48 * function is called by the EM implementation when an error occurs. This
49 * makes it possible to centralize the error management or even modify
50 * some API behavior as an error handler can modify the original API call
51 * return value as well. For instance, a global error handler could modify
52 * the em_send() behavior in such a way that the application never has to
53 * check the return value by catching all errors, fixing them (like
54 * deallocating event on failed send) and returning EM_OK.
55 * - There can be one global error handler and additionally one per each EO.
56 * @{
57 */

85/**
86 * @typedef em_status_t
87 * Error/Status code.
88 *
89 * EM_OK (0) is the general code for success, other values
90 * describe failed operation.
91 * There is a generic error code EM_ERROR, but application should
92 * normally test for not equal to EM_OK.
93 *
94 * @see em_error_handler_t(), em_error()
95 */
96typedef uint32_t em_status_t;
97#define PRI_STAT PRIu32
98#define PRIxSTAT PRIx32
99
100/**
101 * @def EM_OK
102 * Operation successful
103 */
104#define EM_OK 0
105
106/**
107 * @def EM_ERROR
108 * Operation not successful.
109 *
110 * Generic error code, other error codes are system specific.
111 */
112#define EM_ERROR 0xffffffff

175/**
176 * @typedef em_escope_t
177 * Error scope.

179 * Identifies the error scope for interpreting error codes and variable
180 * arguments.

184typedef uint32_t em_escope_t;
185#define PRI_ESCOPE PRIu32
186#define PRIxESCOPE PRIx32

188/**
189 * Error handler prototype
190 *
191 * The error handler is called after EM notices an error or the user has called
192 * em_error().
193 *
194 * The user can register EO specific and/or EM global error handlers. When an
195 * error occurs, EM calls the EO specific error handler, if registered. If
196 * there's no EO specific handler registered or the error occurs outside of an
197 * EO context, EM calls the global error handler.
198 *
199 * The error handler is called with the original error code (that's about to be
200 * returned) from the API call or em_error(). The error scope identifies the
201 * source of the error and how the error code and the variable arguments should
202 * be interpreted (number of arguments and types) in an implementation specific
203 * way.
204 *
205 * @param eo Execution object id
206 * @param error The error code
207 * @param escope Error scope. Identifies the scope for interpreting the
208 * error code and variable arguments.
209 * @param args Variable number and type of arguments
210 *
211 * @return The function may not return depending on implementation, error code
212 * or error scope. If it returns, it can return the original or
213 * modified error code or even EM_OK if it was able to fix the problem.
214 *
215 * @see em_register_error_handler(), em_eo_register_error_handler()
216 */
217typedef em_status_t (*em_error_handler_t)(em_eo_t eo, em_status_t error,
218 em_escope_t escope, va_list args);
219
220/**
221 * Register a global error handler.
222 *
223 * The global error handler is called on EM errors or by em_error() calls,
224 * except when running in an EO context with an EO specific error handler
225 * registered (in which case the EO specific error handler takes precedence).
226 * Note, the provided function will override any previously registered
227 * global error handler.
228 * The EM default global error handler is used when no user provided global
229 * error handler is registered.
230 *
231 * @param handler Error handler.
232 *
233 * @return EM_OK if successful.
234 *
235 * @see em_eo_register_error_handler(), em_unregister_error_handler(),
236 * em_error_handler_t()
237 */
238em_status_t em_register_error_handler(em_error_handler_t handler);
239
240/**
241 * Unregister a global error handler.
242 *
243 * Unregisters any previously registered global error handler and
244 * restores the EM default global error handler into use.
245 *
246 * @return EM_OK if successful.
247 *
248 * @see em_register_error_handler()
249 */
250em_status_t em_unregister_error_handler(void);
251
252/**
253 * Report an error.
254 *
255 * Reported errors are handled by the appropriate (EO specific or global)
256 * error handler.
257 *
258 * Depending on the error/scope/implementation, the function call may not
259 * return.
260 *
261 * @param error Error code
262 * @param escope Error scope. Identifies the scope for interpreting the
263 * error code and the variable arguments.
264 * @param ... Variable number and type of arguments
265 *
266 * @see em_register_error_handler(), em_error_handler_t()
267 */
268void em_error(em_status_t error, em_escope_t escope, ...);

295/**
296 * All EM error scopes (API and other internal) should have bit 31 set
297 *
298 * NOTE: High bit is RESERVED for EM escopes and should not be
299 * used by the application.
300 */
301#define EM_ESCOPE_BIT (0x80000000u)
302
303/**
304 * EM API functions error scope
305 */
306#define EM_ESCOPE_API_TYPE (0xFFu)
307/**
308 * EM Add-on API functions error scope
309 */
310#define EM_ESCOPE_ADDON_API_TYPE (0xFEu)
311/**
312 * EM Internal (non-public API) functions error scope
313 */
314#define EM_ESCOPE_INTERNAL_TYPE (0xFCu)
315
316/**
317 * Mask selects the high byte of the 32-bit escope
318 */
319#define EM_ESCOPE_MASK (0xFF000000)
320/**
321 * EM API functions error mask
322 */
323#define EM_ESCOPE_API_MASK (EM_ESCOPE_BIT | (EM_ESCOPE_API_TYPE << 24))
324/**
325 * EM Add-on API functions error mask
326 */
327#define EM_ESCOPE_ADDON_API_MASK (EM_ESCOPE_BIT | (EM_ESCOPE_ADDON_API_TYPE << 24))
328/**
329 * EM Internal (non-API) functions error mask
330 */
331#define EM_ESCOPE_INTERNAL_MASK (EM_ESCOPE_BIT | (EM_ESCOPE_INTERNAL_TYPE << 24))
332
333/**
334 * Test if the error scope identifies an EM function - API, add-on API or other internal!
335 */
336#define EM_ESCOPE(escope) (EM_ESCOPE_BIT & (escope))
337/**
338 * Test if the error scope identifies an EM API function
339 */
340#define EM_ESCOPE_API(escope) (((escope) & EM_ESCOPE_MASK) == EM_ESCOPE_API_MASK)
341/**
342 * Test if the error scope identifies an EM API function
343 */
344#define EM_ESCOPE_ADDON_API(escope) (((escope) & EM_ESCOPE_MASK) == EM_ESCOPE_ADDON_API_MASK)
345/**
346 * Test if the error scope identifies an EM Internal function
347 */
348#define EM_ESCOPE_INTERNAL(escope) (((escope) & EM_ESCOPE_MASK) == EM_ESCOPE_INTERNAL_MASK)
