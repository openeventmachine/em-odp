46em_status_t
47em_register_error_handler(em_error_handler_t handler)
48{
49 RETURN_ERROR_IF(!em_shm->error_handler.initialized,
50 EM_ERR_NOT_INITIALIZED, EM_ESCOPE_REGISTER_ERROR_HANDLER,
51 "Error Handling not initialized!");
52
53 odp_ticketlock_lock(&em_shm->error_handler.lock);
54
55 em_shm->error_handler.em_error_handler = handler;
56
57 odp_ticketlock_unlock(&em_shm->error_handler.lock);
58
59 return EM_OK;
60}
61
62em_status_t
63em_unregister_error_handler(void)
64{
65 RETURN_ERROR_IF(!em_shm->error_handler.initialized,
66 EM_ERR_NOT_INITIALIZED, EM_ESCOPE_UNREGISTER_ERROR_HANDLER,
67 "Error Handling not initialized!");
68
69 odp_ticketlock_lock(&em_shm->error_handler.lock);
70
71 em_shm->error_handler.em_error_handler = default_error_handler;
72
73 odp_ticketlock_unlock(&em_shm->error_handler.lock);
74
75 return EM_OK;
76}
